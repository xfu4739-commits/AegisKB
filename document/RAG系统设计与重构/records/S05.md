最终 references scope 查询实际读取 `$.subQuestions[0].references[*].documentId`，并与 `$.executionRequests[0].documentScope` 对账：

~~~sql
SELECT s.exchange_id,
       COUNT(*) AS final_reference_count,
       SUM(CASE WHEN ref.document_id IS NULL THEN 1 ELSE 0 END) AS null_document_id_count,
       SUM(CASE
             WHEN ref.document_id IS NOT NULL
              AND JSON_CONTAINS(
                    JSON_EXTRACT(s.snapshot_json,'$.executionRequests[0].documentScope'),
                    CAST(ref.document_id AS JSON)
                  )=0
             THEN 1 ELSE 0
           END) AS out_of_scope_count,
       JSON_EXTRACT(s.snapshot_json,'$.executionRequests[0].documentScope') AS request_document_scope,
       GROUP_CONCAT(DISTINCT ref.document_id ORDER BY ref.document_id SEPARATOR ',')
         AS final_reference_document_ids
FROM aegis_kb_chat_exchange_trace_stage s
CROSS JOIN JSON_TABLE(
  s.snapshot_json,
  '$.subQuestions[0].references[*]'
  COLUMNS(document_id BIGINT PATH '$.documentId')
) AS ref
WHERE s.exchange_id IN (
  2328858467642654721,
  2328858467642654770,
  2328858467642654820,
  2328858467642654870,
  2328858467642654919
)
  AND s.stage_code='RAG_RETRIEVE'
  AND s.status=1
GROUP BY s.exchange_id,
         JSON_EXTRACT(s.snapshot_json,'$.executionRequests[0].documentScope')
ORDER BY s.exchange_id;
~~~
